# Lo que Goombario —`goombario`, el linter de 65816— marca en el runtime y
# está bien. Lo corre pruebas/herramientas/test_linter.py sobre cada
# cartucho armado; su README dice qué revisa cada chequeo y el formato:
# el chequeo, la rutina y el texto de la línea, sin el comentario. Cada una
# lleva arriba el porqué, y una que ya no tapa nada hace fallar la prueba.
#
# Un `pla` seguido de un `beq` es también la forma más corta de preguntar si
# lo que se sacó es cero: la trampa de AGENTS.md ("La trampa que más veces
# mordió") es cuando el salto quería mirar lo de antes del `pl*`. En todas
# las de abajo el salto pregunta por lo que se sacó, y cada una dice qué es.

# El tipo de comparación, guardado antes de leer el segundo operando: 0 es
# "igual", y el resto va a `if_var_other`.
pull-branch op_if_var bne if_var_other
# El valor a escribir: 0 apaga el interruptor.
pull-branch set_switch beq off
# El modo de la pantalla: 0 es cargar, y el cursor empieza en la primera.
pull-branch slots_screen beq first
# Las voces que terminan este cuadro, juntadas en la pila: ninguna, nada.
pull-branch se_after beq ready
# Lo mismo, de las voces 0 a 5.
pull-branch se_extras_after beq none
# sx, la distancia en x, guardada antes de calcular sy: negativa es que el
# evento está a la izquierda del héroe.
pull-branch turn_fire bmi to_right
# El `php`/`plp` guarda el Z de `can_fall` por encima de los dos `stz`:
# es a propósito, y el salto mira la respuesta de `can_fall`.
pull-branch build_hero_jump bne se_can
# Lo dice su comentario: la espera de cero cuadros sigue leyendo en el mismo.
pull-branch music_frame beq continues
# Lo que tenía equipado esa ranura: 0 es que no tenía nada.
pull-branch actor_equip beq taken
# La cantidad que se mueve, guardada por encima de `eq_inv_point`: 0, nada.
pull-branch eq_inv_move beq remain
# La fase, otra vez: la segunda mitad de la vuelta es la del seno negativo.
pull-branch big_wave_table bmi left
# El valor a escribir, otra vez: negativo lleva el signo.
pull-branch msg_letter bpl no_sign

# `close` es la salida que comparten las dos ramas de `slots_screen`, y su
# comentario dice la pila de arriba abajo: su propia dirección de retorno, el
# `slot` y el banco que empujó `slots_screen`. Saca la dirección, saca los
# dos de abajo y la vuelve a poner: los dos `rts` de `slots_screen` salen
# con la pila en orden aunque, rutina por rutina, una deja y la otra saca.
stack slots_screen rts
stack slots_screen plx
